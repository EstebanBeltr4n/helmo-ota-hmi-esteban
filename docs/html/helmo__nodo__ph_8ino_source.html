216/**
217 * @brief Convierte un texto "M.m.p" en sus tres componentes.
218 * @param s texto de la versión
219 * @param[out] M mayor @param[out] m menor @param[out] p parche
220 * @return true si el formato es válido (cada parte entre 0 y 255).

222bool parsearVersion(const char* s, uint8_t& M, uint8_t& m, uint8_t& p) {
223 int a, b, c;
224 if (!s || sscanf(s, "%d.%d.%d", &a, &b, &c) != 3) return false;
225 if (a < 0 || a > 255 || b < 0 || b > 255 || c < 0 || c > 255) return false;
226 M = a; m = b; p = c;
227 return true;
228}

230/** @brief Calcula el SHA-256 de un texto. @param texto entrada @param[out] salida 32 bytes del hash. */
231void sha256Texto(const String& texto, uint8_t salida[32]) {
232 mbedtls_sha256_context ctx;
233 mbedtls_sha256_init(&ctx);
234 mbedtls_sha256_starts(&ctx, 0);
235 mbedtls_sha256_update(&ctx, (const unsigned char*)texto.c_str(), texto.length());
236 mbedtls_sha256_finish(&ctx, salida);
237 mbedtls_sha256_free(&ctx);
238}

255/* Verifica la firma ECDSA P-256 (DER en base64) del mensaje canónico del manifiesto */
256/**
257 * @brief Verifica la firma ECDSA P-256 del manifiesto con la clave pública de HELMO.
258 * @param mensaje texto canónico "HELMO-OTA|producto|versión|tamaño|sha256"
259 * @param firmaB64 firma DER codificada en base64
260 * @return true solo si la firma es auténtica.
261 */
262bool verificarFirma(const String& mensaje, const char* firmaB64) {
263 if (!firmaB64) return false;
264 uint8_t firma[128];
265 size_t lenFirma = 0;
266 if (mbedtls_base64_decode(firma, sizeof(firma), &lenFirma,
267 (const unsigned char*)firmaB64, strlen(firmaB64)) != 0) return false;
268
269 uint8_t hash[32];
270 sha256Texto(mensaje, hash);
271
272 mbedtls_pk_context pk;
273 mbedtls_pk_init(&pk);
274 int r = mbedtls_pk_parse_public_key(&pk, (const unsigned char*)OTA_CLAVE_PUBLICA_PEM,
275 strlen(OTA_CLAVE_PUBLICA_PEM) + 1);
276 if (r != 0) {
277 Serial.printf("[OTA] Clave pública inválida (-0x%04X). ¿Generaste clave_publica_ota.h?\n", -r);
278 } else {
279 r = mbedtls_pk_verify(&pk, MBEDTLS_MD_SHA256, hash, sizeof(hash), firma, lenFirma);
280 }
281 mbedtls_pk_free(&pk);
282 return r == 0;
283}

335/**
336 * @brief Interpreta un mensaje del HMI (HOLA, PING o EVT) y ejecuta la acción de REGLAS_HMI.
337 * @param linea mensaje sin salto de línea; se modifica al separar los campos.
338 */
339void procesarLineaHMI(char* linea) {
340 ultimoMensajeHMIms = millis();
341 char* campo[5] = { nullptr };
342 uint8_t n = 0;
343 for (char* t = strtok(linea, ","); t && n < 5; t = strtok(nullptr, ",")) campo[n++] = t;
344 if (n == 0) return;
345
346 if (!strcmp(campo[0], "HOLA")) {
347 Serial.printf("[HMI] Módulo conectado (v%s)\n", n > 1 ? campo[1] : "?");
348 enviarPHaHMI();
349 return;
350 }
351 if (!strcmp(campo[0], "PING")) return;
352 if (strcmp(campo[0], "EVT") || n < 4) return;
353
354 uint8_t tipo = (campo[1][0] == 'V') ? 1 : (campo[1][0] == 'G') ? 2 : 0;
355 int conf = constrain(atoi(campo[3]), 0, 100);
356
357 static unsigned long ultimoEventoMs = 0;
358 static uint8_t ultimoCodigo = 0, ultimoTipo = 0;
359
360 for (uint8_t i = 0; i < N_REGLAS; i++) {
361 const ReglaHMI& r = REGLAS_HMI[i];
362 if (r.tipo != tipo || strcasecmp(r.etiqueta, campo[2]) != 0) continue;
363
364 // antirrebote: el mismo evento dentro de 3 s se ignora
365 if (r.codigo == ultimoCodigo && r.tipo == ultimoTipo && millis() - ultimoEventoMs < 3000) return;
366 ultimoEventoMs = millis(); ultimoCodigo = r.codigo; ultimoTipo = r.tipo;
367
368 Serial.printf("[HMI] %s '%s' (%d%%)\n", tipo == 1 ? "Voz" : "Gesto", r.etiqueta, conf);
369 enviarHMI(String("ACK,") + r.etiqueta);
370
371 switch (r.accion) {
372 case ACC_MEDIR: phForzado = true; break;
373 case ACC_RECONOCER: if (estadoPH == 2) alertaReconocida = true; enviarPHaHMI(); break;
374 case ACC_MOSTRAR: enviarPHaHMI(); break;
375 case ACC_INCIDENTE: break;
376 }
377 if (r.uplink) encolarEventoHMI(r.tipo, r.codigo, conf, r.accion);
378 return;
379 }
380 Serial.printf("[HMI] Etiqueta sin regla: %s\n", campo[2]);
381}

402/** @brief Toma 10 lecturas del ADC, descarta los extremos y promedia las 6 centrales. @return voltaje en V. */
403float leerVoltajePHFiltrado() {
404 const int N = 10;
405 int m[N];
406 for (int i = 0; i < N; i++) { m[i] = analogRead(ADC_PH_PIN); delay(20); }
407 for (int i = 0; i < N - 1; i++)
408 for (int j = i + 1; j < N; j++)
409 if (m[i] > m[j]) { int t = m[i]; m[i] = m[j]; m[j] = t; }
410 long suma = 0;
411 for (int i = 2; i < 8; i++) suma += m[i];
412 return (suma / 6.0) * (3.3 / 4095.0);
413}

415/** @brief Clasifica el pH según la norma de agua potable. @return 0 normal, 1 aviso, 2 alarma. */
416uint8_t clasificarPH(float ph) {
417 if (ph < PH_LIMITE_MIN || ph > PH_LIMITE_MAX) return 2;
418 if (ph < PH_LIMITE_MIN + PH_MARGEN_AVISO || ph > PH_LIMITE_MAX - PH_MARGEN_AVISO) return 1;
419 return 0;
420}

422/** @brief Mide el pH, actualiza el semáforo y arma el payload del puerto 3. */
423void prepararUplinkPH() {
424 energizarSensorPH(true);
425 float voltaje = leerVoltajePHFiltrado();
426 energizarSensorPH(false);
427
428 float ph = constrain(PH_M * voltaje + PH_B, 0.0, 14.0);
429 uint8_t nuevoEstado = clasificarPH(ph);
430 if (nuevoEstado == 2 && estadoPH != 2) alertaReconocida = false; // alarma nueva
431 ultimoPH = ph;
432 estadoPH = nuevoEstado;
433
434 uint16_t phInt = (uint16_t)(ph * 100.0 + 0.5);
435 uint16_t mv = (uint16_t)(voltaje * 1000.0 + 0.5);
436 uint8_t flags = (estadoPH & 0x03) | (hmiConectado() ? 0x04 : 0) | (enPrueba ? 0x08 : 0);
437
438 appPort = PUERTO_PH;
439 appDataSize = 9;
440 appData[0] = NODE_ID;
441 appData[1] = phInt >> 8; appData[2] = phInt & 0xFF;
442 appData[3] = mv >> 8; appData[4] = mv & 0xFF;
443 appData[5] = FW_MAJOR; appData[6] = FW_MINOR; appData[7] = FW_PATCH;
444 appData[8] = flags;
445
446 Serial.printf("[pH] v%d.%d.%d | pH %.2f | %.3f V | estado %d%s\n", FW_MAJOR, FW_MINOR, FW_PATCH,
447 ph, voltaje, estadoPH, hmiConectado() ? " | HMI" : "");
448 enviarPHaHMI();
449}

607/* Devuelve un código E_*; si todo sale bien reinicia y no retorna */
608/**
609 * @brief Descarga, verifica e instala una imagen firmada.
610 * @details Pasos: manifiesto → firma ECDSA → anti-downgrade → descarga con SHA-256 progresivo
611 * → Update.end() → periodo de prueba → reinicio.
612 * @param trabajo número del trabajo asignado por el servidor
613 * @param M,m,p versión ordenada
614 * @return código E_* de error; si tiene éxito reinicia y no retorna.
615 */
616uint8_t ejecutarOTA(uint8_t trabajo, uint8_t M, uint8_t m, uint8_t p) {
617 if (!conectarWiFiMantenimiento()) return E_WIFI;
618
619 String base = String(OTA_BASE_URL) + "/" + FW_CANAL + "/";
620
621 // --- 1. Manifiesto ---
622 HTTPClient http;
623 if (!iniciarHTTP(http, base + "manifest.json")) return E_MANIF_HTTP;
624 int code = http.GET();
625 if (code != HTTP_CODE_OK) { Serial.printf("[OTA] Manifiesto HTTP %d\n", code); http.end(); return E_MANIF_HTTP; }
626 String json = http.getString();
627 http.end();
628
629 DynamicJsonDocument doc(1536);
630 if (deserializeJson(doc, json)) return E_MANIF_JSON;
631 const char* producto = doc["producto"];
632 const char* version = doc["version"];
633 const char* archivo = doc["archivo"];
634 const char* sha = doc["sha256"];
635 const char* firma = doc["firma"];
636 uint32_t tamano = doc["tamano"] | 0;
637 if (!producto || !version || !archivo || !esHex64(sha) || !firma || tamano == 0) return E_MANIF_JSON;
638
639 if (strcmp(producto, FW_PRODUCTO) != 0) return E_PRODUCTO;
640
641 uint8_t vM, vm, vp;
642 if (!parsearVersion(version, vM, vm, vp)) return E_MANIF_JSON;
643 // Anti-downgrade: debe ser exactamente la versión ordenada y mayor que la actual
644 if (versionNum(vM, vm, vp) != versionNum(M, m, p) || versionNum(vM, vm, vp) <= versionActual())
645 return E_VERSION;
646
647 // --- 2. Firma del manifiesto (antes de tocar la flash) ---
648 String shaMin = String(sha); shaMin.toLowerCase();
649 String canonico = String("HELMO-OTA|") + producto + "|" + version + "|" + String(tamano) + "|" + shaMin;
650 if (!verificarFirma(canonico, firma)) {
651 Serial.println("[ALERTA SEGURIDAD] Firma del manifiesto INVÁLIDA. Actualización rechazada.");
652 return E_FIRMA;
653 }
654 Serial.printf("[OTA] Manifiesto firmado OK: %s v%s (%u bytes)\n", producto, version, tamano);
655
656 // --- 3. Descarga con SHA-256 progresivo ---
657 if (!iniciarHTTP(http, base + archivo)) return E_DESCARGA_HTTP;
658 code = http.GET();
659 if (code != HTTP_CODE_OK) { http.end(); return E_DESCARGA_HTTP; }
660 int largo = http.getSize();
661 if (largo > 0 && (uint32_t)largo != tamano) { http.end(); return E_TAMANO; }
662
663 const esp_partition_t* activa = esp_ota_get_running_partition();
664 if (!Update.begin(tamano)) {
665 Serial.printf("[OTA] Update.begin: %s\n", Update.errorString());
666 http.end();
667 return E_TAMANO;
668 }
669
670 mbedtls_sha256_context ctx;
671 mbedtls_sha256_init(&ctx);
672 mbedtls_sha256_starts(&ctx, 0);
673
674 WiFiClient* stream = http.getStreamPtr();
675 static uint8_t buffer[1024];
676 uint32_t recibidos = 0;
677 unsigned long ultimoDato = millis();
678 int siguientePct = 0;
679 bool falloFlash = false;
680
681 while (recibidos < tamano) {
682 size_t disp = stream->available();
683 if (disp > 0) {
684 size_t aLeer = min(min(disp, sizeof(buffer)), (size_t)(tamano - recibidos));
685 int n = stream->readBytes(buffer, aLeer);
686 if (n <= 0) continue;
687 mbedtls_sha256_update(&ctx, buffer, n);
688 if (Update.write(buffer, n) != (size_t)n) { falloFlash = true; break; }
689 recibidos += n;
690 ultimoDato = millis();
691 escribirLED(!estadoLED);
692 int pct = (int)((uint64_t)recibidos * 100 / tamano);
693 if (pct >= siguientePct) { Serial.printf("[OTA] %u/%u (%d%%)\n", recibidos, tamano, pct); siguientePct = (pct / 10 + 1) * 10; }
694 } else {
695 if (!http.connected()) break;
696 if (millis() - ultimoDato > OTA_TIMEOUT_DATOS_MS) { Serial.println("[OTA] TIMEOUT sin datos"); break; }
697 delay(1);
698 }
699 }
700 http.end();
701
702 uint8_t hash[32];
703 mbedtls_sha256_finish(&ctx, hash);
704 mbedtls_sha256_free(&ctx);
705
706 if (falloFlash) { Update.abort(); return E_FLASH; }
707 if (recibidos < tamano) {
708 Update.abort();
709 Serial.printf("[RESILIENCIA] Interrumpida en %u/%u bytes. Partición %s intacta.\n",
710 recibidos, tamano, activa ? activa->label : "?");
711 return E_INTERRUMPIDA;
712 }
713
714 String calculado = aHex(hash, 32);
715 Serial.printf("[CRIPTO] SHA-256 calculado: %s\n[CRIPTO] SHA-256 firmado : %s\n", calculado.c_str(), shaMin.c_str());
716 if (calculado != shaMin) { Update.abort(); Serial.println("[ALERTA SEGURIDAD] Imagen alterada"); return E_SHA256; }
717
718 if (!Update.end(true)) { Serial.printf("[OTA] Update.end: %s\n", Update.errorString()); return E_IMAGEN; }
719
720 // --- 4. Periodo de prueba en el próximo arranque ---
721 prefs.begin("helmo-ota", false);
722 prefs.putBool("prueba", true);
723 prefs.putString("anterior", activa ? activa->label : "");
724 prefs.putUChar("trabajo", trabajo);
725 prefs.putUChar("arranques", 0);
726 prefs.end();
727
728 Serial.printf("[OTA] v%s verificada y marcada para arranque. Reiniciando...\n", version);
729 apagarWiFi();
730 enviarHMI("OTA,0");
731 delay(1000);
732 ESP.restart();
733 return E_NINGUNO; // no se alcanza
734}

816/** @brief Máquina de estados LoRaWAN de la librería Heltec + tareas no bloqueantes (LED, HMI, OTA). */
817void loop() {
818 actualizarLED();
819 atenderHMI();
820
821 if (enPrueba && millis() - pruebaInicioMs > PRUEBA_TIMEOUT_MS) revertirFirmware("No se unió a la red a tiempo");
822
823 bool radioLibre = (deviceState == DEVICE_STATE_SLEEP) && (millis() - ultimoUplinkMs > 8000);
824
825 if (reinicioSolicitado && radioLibre) { Serial.println("[CMD] Reinicio remoto"); delay(200); ESP.restart(); }
826 if (ordenOTA.pendiente && radioLibre && unido) procesarOrdenOTA();
827
828 // Envío anticipado de eventos urgentes respetando la separación mínima
829 if (unido && deviceState == DEVICE_STATE_SLEEP && hayUplinkUrgente() &&
830 millis() - ultimoUplinkMs >= SEPARACION_MIN_TX_MS) {
831 deviceState = DEVICE_STATE_SEND;
832 }
833
834 switch (deviceState) {
835 case DEVICE_STATE_INIT:
836 LoRaWAN.init(loraWanClass, loraWanRegion);
837 break;
838
839 case DEVICE_STATE_JOIN:
840 LoRaWAN.join();
841 break;
842
843 case DEVICE_STATE_SEND:
844 if (!unido) {
845 unido = true; // la librería solo llega aquí tras el Join Accept
846 Serial.println("[LoRa] Unido a la red");
847 if (enPrueba) confirmarFirmware();
848 }
849 if (reporteOTAPendiente) prepararUplinkOTA();
850 else if (nColaHMI > 0) prepararUplinkHMI();
851 else { phForzado = false; prepararUplinkPH(); }
852 LoRaWAN.send();
853 ultimoUplinkMs = millis();
854 deviceState = DEVICE_STATE_CYCLE;
855 break;
856
857 case DEVICE_STATE_CYCLE:
858 txDutyCycleTime = appTxDutyCycle + randr(-APP_TX_DUTYCYCLE_RND, APP_TX_DUTYCYCLE_RND);
859 LoRaWAN.cycle(txDutyCycleTime);
860 deviceState = DEVICE_STATE_SLEEP;
861 break;
862
863 case DEVICE_STATE_SLEEP:
864#if HMI_HABILITADO
865 Radio.IrqProcess(); // la MCU queda despierta para atender el UART del HMI
866#else
867 LoRaWAN.sleep(loraWanClass); // sin HMI: bajo consumo
868#endif
869 break;
870
871 default:
872 deviceState = DEVICE_STATE_INIT;
873 break;
874 }
