33def mensaje_canonico(producto: str, version: str, tamano: int, sha256_hex: str) -> bytes:
34 """! @brief Texto exacto que se firma; debe coincidir con el que arma el nodo.
35 @return bytes "HELMO-OTA|producto|versión|tamaño|sha256"."""
36 return f"HELMO-OTA|{producto}|{version}|{tamano}|{sha256_hex.lower()}".encode()

39def version_tupla(v: str):
40 """! @brief Convierte "M.m.p" en una tupla comparable.
41 @exception ValueError si el formato no es válido."""
42 partes = v.split(".")
43 if len(partes) != 3 or not all(p.isdigit() and 0 <= int(p) <= 255 for p in partes):
44 raise ValueError(f"versión inválida '{v}': usa M.m.p con valores 0-255")
45 return tuple(int(p) for p in partes)

48def header_c(pem: str) -> str:
49 """! @brief Genera el contenido de clave_publica_ota.h a partir de la clave PEM."""
50 lineas = "".join(f' "{l}\\n"\n' for l in pem.strip().splitlines())
51 return (
52 "/**\n * @file clave_publica_ota.h\n"
53 " * @brief Clave pública ECDSA P-256 con la que el nodo verifica las actualizaciones.\n"
54 " * @details Generado por firmar_firmware.py. La clave pública NO es secreta.\n */\n"
55 "#pragma once\n\n/** @brief Clave pública en formato PEM (SubjectPublicKeyInfo). */\n"
56 "static const char OTA_CLAVE_PUBLICA_PEM[] =\n" + lineas.rstrip("\n") + ";\n"
57 )

60def cmd_llaves(a):
61 """! @brief Crea el par de llaves ECDSA P-256 (una sola vez) y, si se pide, el header del nodo."""
62 d = Path(a.dir)
63 d.mkdir(parents=True, exist_ok=True)
64 priv_path, pub_path = d / "ota_privada.pem", d / "ota_publica.pem"
65 if priv_path.exists() and not a.forzar:
66 sys.exit(f"Ya existe {priv_path}. Usa --forzar solo si quieres REEMPLAZARLA "
67 "(los nodos actuales dejarían de aceptar firmware nuevo).")
68 llave = ec.generate_private_key(ec.SECP256R1())
69 priv_path.write_bytes(llave.private_bytes(
70 serialization.Encoding.PEM, serialization.PrivateFormat.PKCS8, serialization.NoEncryption()))
71 os.chmod(priv_path, 0o600)
72 pub_pem = llave.public_key().public_bytes(
73 serialization.Encoding.PEM, serialization.PublicFormat.SubjectPublicKeyInfo).decode()
74 pub_path.write_text(pub_pem)
75 print(f"Llave privada: {priv_path} (permisos 600, NO la compartas ni la subas a git)")
76 print(f"Llave pública: {pub_path}")
77 if a.header:
78 Path(a.header).write_text(header_c(pub_pem))
79 print(f"Header para el nodo: {a.header}")
80 else:
81 print("\n--- Pega esto en firmware/helmo_nodo_ph/clave_publica_ota.h ---\n")
82 print(header_c(pub_pem))

85def cmd_firmar(a):
86 """! @brief Firma un .bin, lo publica en el servidor de distribución y escribe manifest.json.
87 @details Rechaza imágenes que no son de aplicación ESP32 y versiones no mayores a la publicada."""
88 binario = Path(a.bin).read_bytes()
89 if not binario or binario[0] != 0xE9:
90 sys.exit("El archivo no parece una imagen de aplicación ESP32 (byte mágico 0xE9). "
91 "Usa el .ino.bin de 'Sketch > Export Compiled Binary', no el .merged.bin.")
92 version_tupla(a.version)
93
94 destino = Path(a.destino) / a.canal
95 destino.mkdir(parents=True, exist_ok=True)
96 manifiesto_path = destino / "manifest.json"
97
98 if manifiesto_path.exists() and not a.forzar:
99 anterior = json.loads(manifiesto_path.read_text())
100 if version_tupla(a.version) <= version_tupla(anterior["version"]):
101 sys.exit(f"La versión {a.version} no es mayor que la publicada ({anterior['version']}). "
102 "Los nodos la rechazarían (anti-downgrade).")
103
104 llave = serialization.load_pem_private_key(Path(a.privada).read_bytes(), password=None)
105 sha = hashlib.sha256(binario).hexdigest()
106 firma = llave.sign(mensaje_canonico(a.producto, a.version, len(binario), sha),
107 ec.ECDSA(hashes.SHA256()))
108
109 archivo = f"{a.producto}-{a.version}.bin"
110 shutil.copyfile(a.bin, destino / archivo)
111 manifiesto = {
112 "producto": a.producto,
113 "version": a.version,
114 "archivo": archivo,
115 "tamano": len(binario),
116 "sha256": sha,
117 "firma": base64.b64encode(firma).decode(),
118 "publicado": datetime.now(timezone.utc).isoformat(timespec="seconds"),
119 "notas": a.notas or "",
120 }
121 texto = json.dumps(manifiesto, indent=2, ensure_ascii=False)
122 (destino / f"manifest-{a.version}.json").write_text(texto) # histórico
123 manifiesto_path.write_text(texto)
124 print(f"Publicado {archivo} ({len(binario)} bytes)\nSHA-256 {sha}\nManifiesto: {manifiesto_path}")

127def cmd_verificar(a):
128 """! @brief Comprueba tamaño, SHA-256 y firma de un manifiesto, igual que lo hace el nodo."""
129 m = json.loads(Path(a.manifiesto).read_text())
130 imagen = Path(a.manifiesto).parent / m["archivo"]
131 datos = imagen.read_bytes()
132 pub = serialization.load_pem_public_key(Path(a.publica).read_bytes())
133 sha = hashlib.sha256(datos).hexdigest()
134 ok = True
135 if len(datos) != m["tamano"]:
136 print("✗ El tamaño no coincide"); ok = False
137 if sha != m["sha256"].lower():
138 print("✗ El SHA-256 de la imagen no coincide con el manifiesto"); ok = False
139 try:
140 pub.verify(base64.b64decode(m["firma"]),
141 mensaje_canonico(m["producto"], m["version"], m["tamano"], m["sha256"]),
142 ec.ECDSA(hashes.SHA256()))
143 print("✓ Firma válida")
144 except InvalidSignature:
145 print("✗ FIRMA INVÁLIDA"); ok = False
146 print(f"{'✓' if ok else '✗'} {m['producto']} v{m['version']} ({m['tamano']} bytes)")
147 sys.exit(0 if ok else 1)
