8 * @details
9 * - **Hardware:** Arduino Nano 33 BLE Sense (Rev1 o Rev2), conectado por UART al nodo Heltec V3.
10 * - **Entorno:** Arduino IDE, placa "Arduino Nano 33 BLE" (paquete Arduino Mbed OS Nano Boards).
11 *
12 * @par Funciones principales
13 * -# **Voz:** reconocimiento continuo con el modelo de Edge Impulse
14 * `speech_recognition_Esteban` (Amarillo, Rojo, Verde).
15 * -# **Gestos:** clasificador de movimiento con el IMU (punch, flex). Usa el TensorFlow Lite Micro
16 * incluido en la librería de Edge Impulse para no tener dos copias de TensorFlow.
17 * -# Envía al nodo cada detección confiable; el nodo decide la acción y si la reporta a TTN.
18 * -# Muestra el pH recibido como **semáforo** en el LED RGB y cada detección con su color durante 1,5 s.
19 *
20 * @par Semáforo del LED RGB
21 * verde = normal · amarillo = cerca del límite · rojo parpadeando = alarma sin reconocer ·
22 * rojo fijo = alarma reconocida · azul parpadeando = OTA en curso · azul tenue = esperando al nodo.
23 *
24 * @par Conexión (lógica de 3,3 V en ambas placas)
25 * @verbatim
26 Nano D1 (TX) -> Heltec GPIO 47 (RX)
27 Nano D0 (RX) <- Heltec GPIO 48 (TX)
28 GND -- GND
29 @endverbatim
30 *
31 * @par Protocolo UART (texto, 115200 baud, una línea por mensaje)
32 * @verbatim
33 Nano -> Heltec: HOLA,<ver> | PING | EVT,V,<etiqueta>,<conf%> | EVT,G,<etiqueta>,<conf%>
34 Heltec -> Nano: PH,<pH x100>,<estado>,<reconocida> | OTA,<1|0> | ACK,<etiqueta>
35 @endverbatim
36 *
37 * @pre Instalar el .zip de Edge Impulse y copiar el `model.h` de gestos con el array
38 * renombrado a `modelo_gestos`. Ajustar IMU_REV2 según la placa.
39 * @see helmo_nodo_ph.ino
40 */

86/** @brief Enciende el LED RGB (activo en bajo) con el color indicado. */
87void ponerColor(Color c) {
88 bool r = (c == ROJO || c == AMARILLO || c == MAGENTA || c == BLANCO);
89 bool g = (c == VERDE || c == AMARILLO || c == CIAN || c == BLANCO);
90 bool b = (c == AZUL || c == CIAN || c == MAGENTA || c == BLANCO);
91 digitalWrite(LEDR, r ? LOW : HIGH);
92 digitalWrite(LEDG, g ? LOW : HIGH);
93 digitalWrite(LEDB, b ? LOW : HIGH);
94}

117/** @brief Muestra un color temporal que tiene prioridad sobre el semáforo. @param c color @param ms duración. */
118void mostrarFeedback(Color c, unsigned long ms = 1500) { colorFeedback = c; feedbackHastaMs = millis() + ms; }

120/** @brief Decide el color del LED: retroalimentación, OTA, espera o semáforo de pH. */
121void actualizarLED() {
122 unsigned long t = millis();
123 if ((long)(feedbackHastaMs - t) > 0) { ponerColor(colorFeedback); return; }
124 if (otaEnNodo) { ponerColor((t / 150) % 2 ? AZUL : APAGADO); return; }
125 bool nodoVivo = ultimoMensajeNodoMs && (t - ultimoMensajeNodoMs < 15UL * 60UL * 1000UL);
126 if (phX100 < 0 || !nodoVivo) { ponerColor((t % 2000) < 80 ? AZUL : APAGADO); return; }
127 switch (estadoPH) {
128 case 0: ponerColor(VERDE); break;
129 case 1: ponerColor(AMARILLO); break;
130 default: ponerColor(alertaReconocida ? ROJO : ((t / 300) % 2 ? ROJO : APAGADO)); break;
131 }
132}

139/**
140 * @brief Envía una detección al nodo por UART con el formato EVT,tipo,etiqueta,confianza.
141 * @param tipo 'V' voz o 'G' gesto
142 * @param etiqueta clase detectada
143 * @param confianza probabilidad entre 0 y 1
144 */
145void enviarEvento(char tipo, const char* etiqueta, float confianza) {
146 int pct = (int)(confianza * 100.0 + 0.5);
147 Serial1.print("EVT,"); Serial1.print(tipo); Serial1.print(',');
148 Serial1.print(etiqueta); Serial1.print(','); Serial1.print(pct); Serial1.print('\n');
149 Serial.print(">>> "); Serial.print(tipo == 'V' ? "VOZ " : "GESTO ");
150 Serial.print(etiqueta); Serial.print(" ("); Serial.print(pct); Serial.println("%) enviado al nodo");
151 ultimoEventoMs = millis();
152 mostrarFeedback(colorDeEtiqueta(etiqueta));
153}

155/** @brief Interpreta los mensajes del nodo (PH, OTA, ACK). @param linea mensaje sin salto de línea. */
156void procesarLineaNodo(char* linea) {
157 ultimoMensajeNodoMs = millis();
158 char* c[4] = { nullptr };
159 uint8_t n = 0;
160 for (char* t = strtok(linea, ","); t && n < 4; t = strtok(nullptr, ",")) c[n++] = t;
161 if (n == 0) return;
162
163 if (!strcmp(c[0], "PH") && n >= 4) {
164 phX100 = atoi(c[1]);
165 estadoPH = atoi(c[2]);
166 alertaReconocida = atoi(c[3]) == 1;
167 Serial.print("[NODO] pH "); Serial.print(phX100 / 100.0, 2);
168 Serial.print(" | estado "); Serial.println(estadoPH);
169 } else if (!strcmp(c[0], "OTA") && n >= 2) {
170 otaEnNodo = atoi(c[1]) == 1;
171 Serial.println(otaEnNodo ? "[NODO] Actualizando firmware..." : "[NODO] OTA terminada");
172 } else if (!strcmp(c[0], "ACK") && n >= 2) {
173 Serial.print("[NODO] Confirmó "); Serial.println(c[1]);
174 }
175}

177/** @brief Lee el UART sin bloquear y envía un PING (o HOLA si aún no hay pH) cada 10 s. */
178void atenderUART() {
179 static char buf[64];
180 static uint8_t len = 0;
181 while (Serial1.available()) {
182 char ch = Serial1.read();
183 if (ch == '\n') { buf[len] = 0; if (len) procesarLineaNodo(buf); len = 0; }
184 else if (ch != '\r' && len < sizeof(buf) - 1) buf[len++] = ch;
185 }
186 static unsigned long ultimoPing = 0;
187 if (millis() - ultimoPing > 10000) {
188 ultimoPing = millis();
189 // mientras no tenga un pH, se vuelve a presentar para que el nodo le envíe el último valor
190 Serial1.print(phX100 < 0 ? "HOLA," HMI_VERSION "\n" : "PING\n");
191 }
192}

195 * 4. VOZ (Edge Impulse, audio continuo con doble buffer)
196 * ----------------------------------------------------------------------------------------- */
197/** @brief Doble buffer de audio para la inferencia continua de Edge Impulse. */
198typedef struct {
199 signed short* buffers[2];
200 volatile unsigned char buf_select;
201 volatile unsigned char buf_ready;
202 volatile unsigned int buf_count;
203 unsigned int n_samples;
204} inference_t;
205
206static inference_t inference;
207static volatile bool record_ready = false;
208static signed short* sampleBuffer;
209static int print_results = -(EI_CLASSIFIER_SLICES_PER_MODEL_WINDOW);
210
211/** @brief Callback del micrófono PDM: copia muestras al buffer activo y alterna al llenarse. */
212static void pdm_data_ready_inference_callback(void) {
213 int bytesAvailable = PDM.available();
214 int bytesRead = PDM.read((char*)&sampleBuffer[0], bytesAvailable);
215 if (!record_ready) return;
216 for (int i = 0; i < (bytesRead >> 1); i++) {
217 inference.buffers[inference.buf_select][inference.buf_count++] = sampleBuffer[i];
218 if (inference.buf_count >= inference.n_samples) {
219 inference.buf_select ^= 1;
220 inference.buf_count = 0;
221 inference.buf_ready = 1;
222 }
223 }
224}
225
226/** @brief Reserva los buffers de audio y arranca el micrófono a la frecuencia del modelo. */
227static bool microphone_inference_start(uint32_t n_samples) {
228 inference.buffers[0] = (signed short*)malloc(n_samples * sizeof(signed short));
229 inference.buffers[1] = (signed short*)malloc(n_samples * sizeof(signed short));
230 sampleBuffer = (signed short*)malloc((n_samples >> 1) * sizeof(signed short));
231 if (!inference.buffers[0] || !inference.buffers[1] || !sampleBuffer) return false;
232
233 inference.buf_select = 0;
234 inference.buf_count = 0;
235 inference.n_samples = n_samples;
236 inference.buf_ready = 0;
237
238 PDM.onReceive(&pdm_data_ready_inference_callback);
239 PDM.setBufferSize((n_samples >> 1) * sizeof(int16_t));
240 if (!PDM.begin(1, EI_CLASSIFIER_FREQUENCY)) { Serial.println("ERR: no arrancó el micrófono PDM"); return false; }
241 PDM.setGain(127);
242 record_ready = true;
243 return true;
244}

258/** @brief Clasifica el último trozo de audio y envía la palabra si supera UMBRAL_VOZ. */
259void procesarAudio() {
260 signal_t signal;
261 signal.total_length = EI_CLASSIFIER_SLICE_SIZE;
262 signal.get_data = &microphone_audio_signal_get_data;
263 ei_impulse_result_t result = { 0 };
264
265 if (run_classifier_continuous(&signal, &result, false) != EI_IMPULSE_OK) return;
266
267 size_t mejor = 0;
268 for (size_t i = 1; i < EI_CLASSIFIER_LABEL_COUNT; i++)
269 if (result.classification[i].value > result.classification[mejor].value) mejor = i;
270
271 if (++print_results >= EI_CLASSIFIER_SLICES_PER_MODEL_WINDOW) {
272 Serial.print("[VOZ]");
273 for (size_t i = 0; i < EI_CLASSIFIER_LABEL_COUNT; i++) {
274 Serial.print(' '); Serial.print(result.classification[i].label);
275 Serial.print('='); Serial.print(result.classification[i].value, 2);
276 }
277 Serial.println();
278 print_results = 0;
279 }
280
281 const char* etiqueta = result.classification[mejor].label;
282 float conf = result.classification[mejor].value;
283 if (conf >= UMBRAL_VOZ && !etiquetaIgnorada(etiqueta) && millis() - ultimoEventoMs > ENFRIAMIENTO_MS) {
284 enviarEvento('V', etiqueta, conf);
285 }
286}

288/* -----------------------------------------------------------------------------------------
289 * 5. GESTOS (TFLite Micro + IMU, sin bloquear el audio)
290 * ----------------------------------------------------------------------------------------- */
291constexpr int TAM_ARENA_GESTOS = 8 * 1024;
292alignas(16) static uint8_t arenaGestos[TAM_ARENA_GESTOS];
293static tflite::MicroMutableOpResolver<4> resolverGestos;
294static tflite::MicroInterpreter* interpreteGestos = nullptr;
295static TfLiteTensor* entradaGestos = nullptr;
296static TfLiteTensor* salidaGestos = nullptr;
297bool gestosListos = false;
298int muestrasLeidas = MUESTRAS_GESTO; // == MUESTRAS_GESTO -> esperando movimiento
299
300/** @brief Carga el modelo de gestos en TFLite Micro y valida su entrada (119 x 6 floats). */
301bool iniciarGestos() {
302 const tflite::Model* modelo = tflite::GetModel(modelo_gestos);
303 if (modelo->version() != 3) { Serial.println("ERR: versión de esquema del modelo de gestos"); return false; }
304
305 resolverGestos.AddFullyConnected();
306 resolverGestos.AddSoftmax();
307 resolverGestos.AddRelu();
308 resolverGestos.AddReshape();
309
310 static tflite::MicroInterpreter interprete(modelo, resolverGestos, arenaGestos, TAM_ARENA_GESTOS);
311 interpreteGestos = &interprete;
312 if (interpreteGestos->AllocateTensors() != kTfLiteOk) {
313 Serial.println("ERR: AllocateTensors (sube TAM_ARENA_GESTOS a 16 o 32 KB)");
314 return false;
315 }
316 entradaGestos = interpreteGestos->input(0);
317 salidaGestos = interpreteGestos->output(0);
318 if (entradaGestos->type != kTfLiteFloat32 || entradaGestos->bytes != MUESTRAS_GESTO * 6 * sizeof(float)) {
319 Serial.println("ERR: el modelo de gestos no espera 119x6 floats");
320 return false;
321 }
322 return true;
323}

325/** @brief Ejecuta el modelo de gestos y envía el gesto si supera UMBRAL_GESTO. */
326void clasificarGesto() {
327 if (interpreteGestos->Invoke() != kTfLiteOk) { Serial.println("ERR: Invoke gestos"); return; }
328 size_t mejor = 0;
329 Serial.print("[GESTO]");
330 for (size_t i = 0; i < NUM_GESTOS; i++) {
331 float v = salidaGestos->data.f[i];
332 Serial.print(' '); Serial.print(GESTOS[i]); Serial.print('='); Serial.print(v, 2);
333 if (v > salidaGestos->data.f[mejor]) mejor = i;
334 }
335 Serial.println();
336 float conf = salidaGestos->data.f[mejor];
337 if (conf >= UMBRAL_GESTO && millis() - ultimoEventoMs > ENFRIAMIENTO_MS) enviarEvento('G', GESTOS[mejor], conf);
338}

340/** @brief Espera un movimiento fuerte, toma 119 muestras normalizadas y clasifica, sin bloquear el audio. */
341void atenderIMU() {
342 if (!gestosListos) return;
343 float aX, aY, aZ, gX, gY, gZ;
344
345 if (muestrasLeidas == MUESTRAS_GESTO) { // esperando un movimiento fuerte
346 if (IMU.accelerationAvailable()) {
347 IMU.readAcceleration(aX, aY, aZ);
348 if (fabs(aX) + fabs(aY) + fabs(aZ) >= UMBRAL_MOVIMIENTO_G && millis() - ultimoEventoMs > ENFRIAMIENTO_MS)
349 muestrasLeidas = 0;
350 }
351 return;
352 }
353
354 while (muestrasLeidas < MUESTRAS_GESTO && IMU.accelerationAvailable() && IMU.gyroscopeAvailable()) {
355 IMU.readAcceleration(aX, aY, aZ);
356 IMU.readGyroscope(gX, gY, gZ);
357 float* d = &entradaGestos->data.f[muestrasLeidas * 6]; // misma normalización del entrenamiento
358 d[0] = (aX + 4.0) / 8.0; d[1] = (aY + 4.0) / 8.0; d[2] = (aZ + 4.0) / 8.0;
359 d[3] = (gX + 2000.0) / 4000.0; d[4] = (gY + 2000.0) / 4000.0; d[5] = (gZ + 2000.0) / 4000.0;
360 muestrasLeidas++;
361 }
362 if (muestrasLeidas == MUESTRAS_GESTO) clasificarGesto();
363}
