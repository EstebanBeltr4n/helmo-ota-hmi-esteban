18/** @brief Lee un entero de 16 bits big-endian. */
19function u16(b, i) { return (b[i] << 8) | b[i + 1]; }

21/**
22 * @brief Decodifica los uplinks de los puertos 3 (pH), 4 (HMI), 5 (OTA) y 21 (FUOTA de laboratorio).
23 * @param input objeto de TTN con fPort y bytes
24 * @return {data} con los campos decodificados, o {errors}
25 */
26function decodeUplink(input) {
27 var b = input.bytes;
28 switch (input.fPort) {
29 case 3: {
30 if (b.length < 6) return { errors: ["pH: payload corto"] };
31 var d = { tipo: "lectura_ph", node_id: b[0], ph: u16(b, 1) / 100, voltaje: u16(b, 3) / 1000 };
32 if (b.length >= 9) {
33 d.fw_version = b[5] + "." + b[6] + "." + b[7];
34 d.estado_ph = ESTADOS_PH[b[8] & 0x03];
35 d.hmi_conectado = (b[8] & 0x04) !== 0;
36 d.fw_en_prueba = (b[8] & 0x08) !== 0;
37 } else {
38 d.fw_version = b[5] + ".0.0"; // firmware anterior (6 bytes)
39 }
40 return { data: d };
41 }
42 case 4: {
43 if (b.length < 5) return { errors: ["HMI: payload corto"] };
44 var esVoz = b[1] === 1;
45 return { data: {
46 tipo: "evento_hmi", node_id: b[0], fuente: esVoz ? "voz" : "gesto",
47 etiqueta: (esVoz ? VOZ : GESTO)[b[2]] || ("codigo_" + b[2]),
48 confianza: b[3] / 100, accion: ACCION[b[4]] || ("accion_" + b[4])
49 } };
50 }
51 case 5: {
52 if (b.length < 7) return { errors: ["OTA: payload corto"] };
53 return { data: {
54 tipo: "reporte_ota", node_id: b[0], trabajo: b[1],
55 resultado: RESULTADO_OTA[b[2]] || ("codigo_" + b[2]),
56 fw_version: b[3] + "." + b[4] + "." + b[5],
57 detalle: ERROR_OTA[b[6]] || ("error_" + b[6])
58 } };
59 }
60 case 21: // reportes del protocolo FUOTA de laboratorio (ChirpStack), solo para compatibilidad
61 return { data: { tipo: "fuota_lab", bytes: b } };
62 default:
63 return { warnings: ["puerto sin decodificador: " + input.fPort], data: { bytes: b } };
64 }
65}

67// Downlink desde la consola de TTN, puerto 10. Ejemplos de JSON:
68// {"comando":"ota","trabajo":7,"version":"2.1.0"}
69// {"comando":"intervalo","minutos":10}
70// {"comando":"medir"} {"comando":"reiniciar"}
71/** @brief Convierte un comando JSON (ota, intervalo, medir, reiniciar) en bytes del puerto 10. */
72function encodeDownlink(input) {
73 var d = input.data || {};
74 if (d.comando === "ota") {
75 var v = String(d.version || "").split(".");
76 if (v.length !== 3) return { errors: ["version debe ser M.m.p"] };
77 return { fPort: 10, bytes: [0x01, (d.trabajo || 1) & 0xFF, +v[0] & 0xFF, +v[1] & 0xFF, +v[2] & 0xFF] };
78 }
79 if (d.comando === "intervalo") {
80 var m = d.minutos | 0;
81 if (m < 1 || m > 1440) return { errors: ["minutos entre 1 y 1440"] };
82 return { fPort: 10, bytes: [0x02, (m >> 8) & 0xFF, m & 0xFF] };
83 }
84 if (d.comando === "medir") return { fPort: 10, bytes: [0x03] };
85 if (d.comando === "reiniciar") return { fPort: 10, bytes: [0x04] };
86 return { errors: ["comando desconocido"] };
87}

89/** @brief Muestra en la consola de TTN el comando de un downlink del puerto 10. */
90function decodeDownlink(input) {
91 var b = input.bytes;
92 switch (b[0]) {
93 case 0x01: return { data: { comando: "ota", trabajo: b[1], version: b[2] + "." + b[3] + "." + b[4] } };
94 case 0x02: return { data: { comando: "intervalo", minutos: u16(b, 1) } };
95 case 0x03: return { data: { comando: "medir" } };
96 case 0x04: return { data: { comando: "reiniciar" } };
97 default: return { errors: ["comando desconocido"] };
98 }
