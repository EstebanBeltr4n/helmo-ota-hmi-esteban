79 """! @brief Crea un cliente MQTT con TLS hacia TTN (compatible con paho-mqtt 1.x y 2.x)."""

81 sys.exit("Faltan TTN_APP_ID / TTN_API_KEY (archivo .env)")
82 cid = f"{nombre}-{os.getpid()}"

92def enviar_downlink(dev_id, payload: bytes, prioridad="NORMAL"):
93 """! @brief Encola un downlink en TTN por el puerto 10.
94 @param prioridad prioridad del downlink en TTN (NORMAL por defecto)
95 @param dev_id End device ID en TTN
96 @param payload bytes del comando
97 @details En Clase A el nodo lo recibe después de su próximo uplink."""
98 topic = f"v3/{CFG['app_id']}@ttn/devices/{dev_id}/down/push"
99 cuerpo = json.dumps({"downlinks": [{
100 "f_port": PUERTO_CMD,
101 "frm_payload": base64.b64encode(payload).decode(),
102 "priority": prioridad,
103 }]})
104 c = cliente_mqtt("helmo-ota-cmd")
105 c.connect(CFG["host"], 8883)
106 c.loop_start()
107 info = c.publish(topic, cuerpo, qos=1)
108 info.wait_for_publish(timeout=15)
109 c.loop_stop()
110 c.disconnect()
111 if not info.is_published():
112 sys.exit("No se pudo publicar el downlink en TTN")
113 print(f"Downlink encolado para {dev_id}: {payload.hex()} (llega tras su próximo uplink, Clase A)")

134def ordenar(dev_id, version, d=None):
135 """! @brief Ordena a un nodo actualizarse a la versión publicada y registra el trabajo en MongoDB."""
136 d = d or db()
137 m = manifiesto_publicado()
138 if version != m["version"]:
139 sys.exit(f"La versión publicada es {m['version']}; el nodo rechazaría {version}.")
140 nodo = d.nodos.find_one({"_id": dev_id})
141 if nodo and nodo.get("fw_version") and version_tupla(nodo["fw_version"]) >= version_tupla(version):
142 print(f"{dev_id} ya tiene {nodo['fw_version']}; no se ordena."); return
143 if d.ota_trabajos.find_one({"dev_id": dev_id, "estado": "ORDENADO"}):
144 sys.exit(f"{dev_id} ya tiene un trabajo OTA en curso. Espera su reporte o que venza.")
145
146 trabajo = siguiente_trabajo(d)
147 M, mi, p = version_tupla(version)
148 enviar_downlink(dev_id, bytes([0x01, trabajo, M, mi, p]))
149 d.ota_trabajos.insert_one({
150 "dev_id": dev_id, "trabajo": trabajo, "version_objetivo": version,
151 "version_inicial": (nodo or {}).get("fw_version"), "estado": "ORDENADO",
152 "creado": ahora(), "actualizado": ahora(), "historial": [],
153 })
154 print(f"Trabajo OTA {trabajo}: {dev_id} -> v{version}")

157def campana(max_simultaneos):
158 """! @brief Ordena la versión publicada a los nodos desactualizados, sin superar max_simultaneos."""
159 d = db()
160 m = manifiesto_publicado()
161 en_curso = d.ota_trabajos.count_documents({"estado": "ORDENADO"})
162 cupo = max(0, max_simultaneos - en_curso)
163 pendientes = [n for n in d.nodos.find({"fw_version": {"$exists": True}})
164 if version_tupla(n["fw_version"]) < version_tupla(m["version"])
165 and not d.ota_trabajos.find_one({"dev_id": n["_id"], "estado": "ORDENADO"})]
166 print(f"Versión publicada {m['version']} | desactualizados {len(pendientes)} | cupo {cupo}")
167 for n in pendientes[:cupo]:
168 ordenar(n["_id"], m["version"], d)

174def procesar_uplink(d, msg):
175 """! @brief Registra en MongoDB la versión del nodo, los eventos HMI y los reportes OTA de un uplink."""
176 dev_id = msg.get("end_device_ids", {}).get("device_id")
177 up = msg.get("uplink_message", {})
178 datos = up.get("decoded_payload") or {}
179 tipo = datos.get("tipo")
180 if not dev_id or not tipo:
181 return
182 recibido = msg.get("received_at") or ahora().isoformat()
183
184 cambios = {"ultima_vista": ahora(), "ultimo_puerto": up.get("f_port")}
185 if "fw_version" in datos and tipo == "lectura_ph":
186 cambios.update({k: datos[k] for k in ("fw_version", "estado_ph", "hmi_conectado", "fw_en_prueba", "ph")
187 if k in datos})
188 d.nodos.update_one({"_id": dev_id}, {"$set": cambios, "$setOnInsert": {"creado": ahora()}}, upsert=True)
189
190 if tipo == "evento_hmi":
191 d.hmi_eventos.insert_one({"dev_id": dev_id, "recibido": recibido, **datos})
192 print(f"[HMI] {dev_id}: {datos.get('fuente')} '{datos.get('etiqueta')}' -> {datos.get('accion')}")
193
194 elif tipo == "reporte_ota":
195 resultado = datos.get("resultado")
196 d.nodos.update_one({"_id": dev_id}, {"$set": {"fw_version": datos.get("fw_version")}})
197 t = d.ota_trabajos.find_one_and_update(
198 {"dev_id": dev_id, "trabajo": datos.get("trabajo")},
199 {"$set": {"estado": resultado, "actualizado": ahora(), "detalle": datos.get("detalle"),
200 "fw_reportada": datos.get("fw_version")},
201 "$push": {"historial": {"en": recibido, **datos}}},
202 sort=[("creado", -1)])
203 print(f"[OTA] {dev_id} trabajo {datos.get('trabajo')}: {resultado} "
204 f"(v{datos.get('fw_version')}, {datos.get('detalle')}){'' if t else ' [sin trabajo registrado]'}")

207def vencer_trabajos(d):
208 """! @brief Marca como VENCIDO todo trabajo sin respuesta en TRABAJO_VENCE_H horas."""
209 limite = ahora() - timedelta(hours=TRABAJO_VENCE_H)
210 r = d.ota_trabajos.update_many({"estado": "ORDENADO", "creado": {"$lt": limite}},
211 {"$set": {"estado": "VENCIDO", "actualizado": ahora()}})
212 if r.modified_count:
213 print(f"[OTA] {r.modified_count} trabajo(s) vencido(s)")

216def escuchar():
217 """! @brief Servicio principal: se suscribe a los uplinks de la aplicación en TTN."""
218 d = db()
219 d.ota_trabajos.create_index([("dev_id", 1), ("trabajo", 1)])

223 def al_conectar(cliente, *args):
224 cliente.subscribe(f"v3/{CFG['app_id']}@ttn/devices/+/up", qos=1)
225 print(f"[MQTT] Conectado a {CFG['host']}, escuchando uplinks de {CFG['app_id']}")
226
227 def al_mensaje(cliente, userdata, m):
228 try:
229 procesar_uplink(d, json.loads(m.payload))
230 except Exception as e: # un mensaje malo no debe tumbar el servicio
231 print(f"[ERROR] {e}")
232
233 c.on_connect = al_conectar
234 c.on_message = al_mensaje
235 c.reconnect_delay_set(min_delay=2, max_delay=120)
236 c.connect(CFG["host"], 8883)
237 c.loop_start()
238 try:
239 while True:
240 vencer_trabajos(d)
241 time.sleep(600)
242 except KeyboardInterrupt:
243 c.loop_stop()
